rproduct:
    irmovq  $1, %rax          
    cmpq    $1, %rsi          
    jle     .Ldone            
    
    pushq   %rbx              
    rrmovq    %rdi, %rbx        
    leaq    8(%rdi), %rdi     
    subq    $1, %rsi          
    call    rproduct          
    imulq   (%rbx), %rax      
    popq    %rbx              
.Ldone:
    ret
