/* Bubble sort: Array version */
void bubble_a(long *data, long count) {
	long *p, *last;
	for (last = data + count - 1; last > data; last--) {
		for (p = data; p < last; p++)
			if (*(p + 1) < *p) {
				/* Swap adjacent elements */
				long t = *(p + 1);
				*(p + 1) = *p;
				*p = t;
			}
	}
}
